Función de periodo — todo month_key sale de aquí, no del mes calendario:

create or replace function public.month_key(p_user uuid, p_at timestamptz)
returns text
language plpgsql
stable
security definer
set search_path = public
as $$
declare
  v_tz    text := 'America/Guayaquil';
  v_start smallint := 1;
  v_local date;
begin
  select coalesce(p.timezone,'America/Guayaquil'), coalesce(s.month_start_day,1)
    into v_tz, v_start
  from public.profiles p
  left join public.user_settings s on s.user_id = p.id
  where p.id = p_user;

  v_local := (p_at at time zone v_tz)::date;
  if extract(day from v_local) < v_start then
    v_local := (v_local - interval '1 month')::date;
  end if;
  return to_char(v_local, 'YYYY-MM');
end $$;

Resolución de alias — se llama en el ingest antes de crear la transacción:

create or replace function public.resolve_merchant(p_user uuid, p_raw text)
returns table (merchant_id uuid, category_id uuid, alias_id uuid)
language sql stable as $$
  with needle as (select upper(regexp_replace(coalesce(p_raw,''), '\s+', ' ', 'g')) as v)
  select m.id, coalesce(a.merchant_id is not null, false)::bool and m.default_category_id is not null
           then m.default_category_id else m.default_category_id end, a.id
  from public.merchant_aliases a
  join public.merchants m on m.id = a.merchant_id
  cross join needle n
  where a.user_id = p_user
    and (
      (a.match_type = 'exact'    and n.v = upper(a.raw_pattern)) or
      (a.match_type = 'prefix'   and n.v like upper(a.raw_pattern) || '%') or
      (a.match_type = 'contains' and n.v like '%' || upper(a.raw_pattern) || '%')
    )
  order by (a.source = 'user') desc, a.hit_count desc, length(a.raw_pattern) desc
  limit 1;
$$;

Ojo con esa función: el case que escribí ahí es redundante. La versión correcta del select es:

  select m.id, m.default_category_id, a.id

create or replace function public.tg_tx_month_key()
returns trigger language plpgsql as $$
begin
  new.month_key := public.month_key(new.user_id, new.occurred_at);
  return new;
end $$;

create trigger tx_month_key before insert or update of occurred_at, user_id
  on public.transactions for each row execute function public.tg_tx_month_key();

create trigger tx_updated_at before update on public.transactions
  for each row execute function public.tg_set_updated_at();

-- financed / kind siempre reflejan la existencia del plan.
create or replace function public.tg_sync_financed()
returns trigger language plpgsql as $$
begin
  if tg_op = 'DELETE' then
    update public.transactions
       set financed = false, kind = 'purchase'
     where id = old.transaction_id;
    return old;
  end if;
  update public.transactions
     set financed = true, kind = 'deferred_purchase'
   where id = new.transaction_id;
  return new;
end $$;

create trigger sync_financed after insert or delete on public.installment_plans
  for each row execute function public.tg_sync_financed();

-- paid_count = count(installments where status = 'paid')
create or replace function public.tg_sync_paid_count()
returns trigger language plpgsql as $$
declare v_plan uuid := coalesce(new.installment_plan_id, old.installment_plan_id);
begin
  update public.installment_plans p
     set paid_count = (select count(*) from public.installments i
                        where i.installment_plan_id = v_plan and i.status = 'paid'),
         status = case when (select count(*) from public.installments i
                              where i.installment_plan_id = v_plan and i.status <> 'paid') = 0
                       then 'completed' else p.status end
   where p.id = v_plan;
  return null;
end $$;

create trigger sync_paid_count after insert or update or delete on public.installments
  for each row execute function public.tg_sync_paid_count();

returns public.budgets
language sql stable as $$
  select * from public.budgets
   where user_id = p_user and (month_key = p_month or month_key is null)
   order by month_key nulls last

9 · Vistas derivadas
Nada de esto se persiste. security_invoker = on hace que las vistas respeten el RLS del usuario que consulta.

-- Base común: excluye lo ignorado y lo que no cuenta como gasto.
create view public.v_countable_transactions with (security_invoker = on) as
  select * from public.transactions
   where review_status <> 'ignored' and direction = 'expense';

-- Gasto por categoría: Home + Presupuestos.
create view public.v_category_monthly with (security_invoker = on) as
select t.user_id, t.month_key, t.category_id, c.name, c.key, c.color_token, c.icon,
       sum(t.amount_cents)                      as spent_cents,
       count(*)                                 as transaction_count,
       round(sum(t.amount_cents)::numeric
             / nullif(sum(sum(t.amount_cents)) over (partition by t.user_id, t.month_key),0), 4) as share_of_total,
       cb.limit_cents,
       round(sum(t.amount_cents)::numeric / nullif(cb.limit_cents,0), 4) as usage,
       case
         when cb.limit_cents is null then null
         when sum(t.amount_cents) > cb.limit_cents then 'exceeded'
         when sum(t.amount_cents)::numeric / cb.limit_cents >= b.alert_threshold then 'near_limit'
         else 'normal'
       end as status
  from public.v_countable_transactions t
  join public.categories c on c.id = t.category_id
  left join public.budgets b
         on b.user_id = t.user_id and (b.month_key = t.month_key or b.month_key is null)
  left join public.category_budgets cb
         on cb.budget_id = b.id and cb.category_id = t.category_id and cb.active
 group by t.user_id, t.month_key, t.category_id, c.name, c.key, c.color_token, c.icon,
          cb.limit_cents, b.alert_threshold;

create view public.v_bank_monthly with (security_invoker = on) as

       sum(t.amount_cents) as spent_cents,
       count(*)            as transaction_count,
       round(sum(t.amount_cents)::numeric
             / nullif(sum(sum(t.amount_cents)) over (partition by t.user_id, t.month_key),0), 4) as share_of_total
  from public.v_countable_transactions t
  join public.bank_accounts ba on ba.id = t.bank_account_id
  join public.banks bk on bk.id = ba.bank_id
 group by t.user_id, t.month_key, t.bank_account_id, bk.short_name, bk.monogram, bk.brand_color;

create view public.v_monthly_summary with (security_invoker = on) as
with spend as (
  select user_id, month_key, sum(amount_cents) as total_spent_cents, count(*) as transaction_count
    from public.v_countable_transactions group by user_id, month_key
), fin as (
  select t.user_id, i.due_month_key as month_key, sum(i.amount_cents) as financed_cents
    from public.installments i
    join public.installment_plans p on p.id = i.installment_plan_id
    join public.transactions t on t.id = p.transaction_id
   where t.review_status <> 'ignored'
   group by t.user_id, i.due_month_key
), pend as (
  select user_id, month_key, count(*) as pending_review_count
    from public.transactions where review_status = 'needs_review'
   group by user_id, month_key
)
select s.user_id, s.month_key,
       s.total_spent_cents,
       s.transaction_count,
       coalesce(f.financed_cents, 0) as financed_this_month_cents,
       coalesce(pe.pending_review_count, 0) as pending_review_count,
       b.total_limit_cents,
       -- OJO: "Disponible" del Home. Esto asume tope de presupuesto como techo.
       -- Si en realidad es ingreso mensual, hace falta una tabla que el diseño no tiene.
       b.total_limit_cents - s.total_spent_cents as available_cents,
       round(s.total_spent_cents::numeric
             / nullif(lag(s.total_spent_cents) over (partition by s.user_id order by s.month_key),0) - 1, 4)
         as trend_vs_previous_month
  from spend s
  left join fin  f  on f.user_id  = s.user_id and f.month_key  = s.month_key
  left join pend pe on pe.user_id = s.user_id and pe.month_key = s.month_key
  left join public.budgets b
         on b.user_id = s.user_id and (b.month_key = s.month_key or b.month_key is null);

create view public.v_review_queue with (security_invoker = on) as
select t.id as transaction_id, t.user_id, t.display_label, t.merchant_raw,
       t.amount_cents, t.occurred_at, t.card_id, t.bank_account_id,
       n.id as notification_id, n.body, n.source_label, n.parse_confidence, n.missing_fields,
       case
         when t.category_id is null and t.merchant_id is null then 'unknown_merchant'
         when t.category_id is null                            then 'unknown_category'
         when coalesce(n.parse_confidence,0) < 0.7             then 'low_confidence'
         else 'missing_amount'
       end as reason
  from public.transactions t
  left join public.captured_notifications n on n.id = t.captured_notification_id
 where t.review_status = 'needs_review';

create or replace function public.budget_pace(p_user uuid, p_month char(7))
returns table (day_of_period int, expected_spent_cents bigint,
               actual_spent_cents bigint, pace_deviation numeric, over_pace boolean)
language plpgsql stable as $$
declare
  v_start smallint; v_tz text; v_first date; v_next date; v_today date;
  v_limit bigint; v_spent bigint; v_days int; v_elapsed int;
begin
  select coalesce(s.month_start_day,1), coalesce(p.timezone,'America/Guayaquil')
    into v_start, v_tz
    from public.profiles p left join public.user_settings s on s.user_id = p.id
   where p.id = p_user;

  v_first  := to_date(p_month,'YYYY-MM') + (v_start - 1);
  v_next   := (v_first + interval '1 month')::date;
  v_today  := (now() at time zone v_tz)::date;
  v_days   := v_next - v_first;
  v_elapsed:= greatest(1, least(v_days, v_today - v_first + 1));

  select total_limit_cents into v_limit from public.effective_budget(p_user, p_month);
  select coalesce(sum(amount_cents),0) into v_spent
    from public.v_countable_transactions where user_id = p_user and month_key = p_month;

  expected_spent_cents := (v_limit::numeric * v_elapsed / v_days)::bigint;
  actual_spent_cents   := v_spent;
  day_of_period        := v_elapsed;
  pace_deviation       := round(v_spent::numeric / nullif(expected_spent_cents,0) - 1, 4);
  over_pace            := pace_deviation > 0;
  return next;
end $$;

create or replace function public.tg_on_auth_user_created()
returns trigger language plpgsql security definer set search_path = public as $$
begin
  insert into public.profiles (id, display_name)
    values (new.id, coalesce(new.raw_user_meta_data->>'full_name', split_part(new.email,'@',1)));
  insert into public.user_settings (user_id) values (new.id);
  insert into public.capture_permissions (user_id, platform) values (new.id,'ios'),(new.id,'android');
  return new;
end $$;
